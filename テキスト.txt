<script>
let editMode='pen',penColor='#f0a500',penSize=4,textColor='#ff4757';
let isDrawing=false,cropStart=null,cropRect=null,drawHistory=[];
let albums=loadAlbums(),selectedOutputImages=[],selectedTextNode=null,textNodes=[];

function switchTab(name){
  document.querySelectorAll('.tab').forEach((t,i)=>t.classList.toggle('active',['edit','album','output'][i]===name));
  document.querySelectorAll('.panel').forEach(p=>p.classList.remove('active'));
  document.getElementById('panel-'+name).classList.add('active');
  if(name==='album')renderAlbums();
  if(name==='output')renderOutputPicker();
}

const canvas=document.getElementById('main-canvas');
const ctx=canvas.getContext('2d');

function loadImageFile(e){
  const file=e.target.files[0];if(!file)return;
  const reader=new FileReader();
  reader.onload=ev=>{const img=new Image();img.onload=()=>setupCanvas(img);img.src=ev.target.result;};
  reader.readAsDataURL(file);e.target.value='';
}

function setupCanvas(img){
  const MAX=1400;let w=img.naturalWidth,h=img.naturalHeight;
  if(w>MAX){h=Math.round(h*MAX/w);w=MAX;}
  if(h>MAX){w=Math.round(w*MAX/h);h=MAX;}
  canvas.width=w;canvas.height=h;
  ctx.drawImage(img,0,0,w,h);
  document.getElementById('edit-no-image').style.display='none';
  clearTexts();drawHistory=[];saveHistory();fitCanvas();
}

function fitCanvas(){
  const wrap=document.getElementById('canvas-wrap');
  const wr=wrap.getBoundingClientRect();
  const scale=Math.min(wr.width/canvas.width,wr.height/canvas.height,1);
  canvas.style.width=(canvas.width*scale)+'px';
  canvas.style.height=(canvas.height*scale)+'px';
  const tl=document.getElementById('text-layer');
  tl.style.width=canvas.style.width;tl.style.height=canvas.style.height;
  const cr=canvas.getBoundingClientRect();const wr2=wrap.getBoundingClientRect();
  tl.style.left=(cr.left-wr2.left)+'px';tl.style.top=(cr.top-wr2.top)+'px';
}
window.addEventListener('resize',fitCanvas);

function canvasPoint(e){
  const r=canvas.getBoundingClientRect();
  const sx=canvas.width/r.width,sy=canvas.height/r.height;
  const cx=e.touches?e.touches[0].clientX:e.clientX;
  const cy=e.touches?e.touches[0].clientY:e.clientY;
  return{x:(cx-r.left)*sx,y:(cy-r.top)*sy};
}

canvas.addEventListener('mousedown',startDraw);
canvas.addEventListener('mousemove',draw);
canvas.addEventListener('mouseup',endDraw);
canvas.addEventListener('touchstart',e=>{e.preventDefault();startDraw(e);},{passive:false});
canvas.addEventListener('touchmove',e=>{e.preventDefault();draw(e);},{passive:false});
canvas.addEventListener('touchend',e=>{e.preventDefault();endDraw(e);},{passive:false});

function startDraw(e){
  if(!canvas.width||editMode==='text')return;
  isDrawing=true;const p=canvasPoint(e);
  if(editMode==='crop'){cropStart=p;document.getElementById('crop-overlay').style.display='block';return;}
  saveHistory();ctx.beginPath();ctx.moveTo(p.x,p.y);
  if(editMode==='highlight'){ctx.globalAlpha=0.4;ctx.globalCompositeOperation='source-over';}
  else if(editMode==='erase'){ctx.globalCompositeOperation='destination-out';ctx.globalAlpha=1;}
  else{ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';}
  ctx.strokeStyle=penColor;ctx.lineWidth=editMode==='highlight'?penSize*3:penSize;
  ctx.lineCap='round';ctx.lineJoin='round';
}
function draw(e){
  if(!isDrawing||!canvas.width||editMode==='text')return;
  const p=canvasPoint(e);
  if(editMode==='crop'){updateCropOverlay(cropStart,p);cropRect={x:Math.min(cropStart.x,p.x),y:Math.min(cropStart.y,p.y),w:Math.abs(p.x-cropStart.x),h:Math.abs(p.y-cropStart.y)};return;}
  ctx.lineTo(p.x,p.y);ctx.stroke();
}
function endDraw(){isDrawing=false;ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';}

function updateCropOverlay(a,b){
  const r=canvas.getBoundingClientRect();const wrap=document.getElementById('canvas-wrap').getBoundingClientRect();
  const sx=r.width/canvas.width,sy=r.height/canvas.height;
  const ov=document.getElementById('crop-overlay');
  ov.style.left=(Math.min(a.x,b.x)*sx+r.left-wrap.left)+'px';
  ov.style.top=(Math.min(a.y,b.y)*sy+r.top-wrap.top)+'px';
  ov.style.width=Math.abs(a.x-b.x)*sx+'px';ov.style.height=Math.abs(a.y-b.y)*sy+'px';
}

function applyCrop(){
  if(!cropRect||cropRect.w<5||cropRect.h<5){toast('トリミング範囲を指定してください');return;}
  const{x,y,w,h}=cropRect;const imgData=ctx.getImageData(x,y,w,h);
  canvas.width=w;canvas.height=h;ctx.putImageData(imgData,0,0);
  document.getElementById('crop-overlay').style.display='none';
  cropRect=null;cropStart=null;fitCanvas();saveHistory();toast('✂️ トリミング完了');
}

function saveHistory(){if(!canvas.width)return;drawHistory.push(canvas.toDataURL());if(drawHistory.length>20)drawHistory.shift();}
function undoDraw(){
  if(drawHistory.length<=1){toast('これ以上戻れません');return;}

// TEXT
function placeText(){
  if(!canvas.width){toast('先に画像を開いてください');return;}
  const content=document.getElementById('text-content').value.trim();
  if(!content){toast('テキストを入力してください');return;}
  const size=parseInt(document.getElementById('text-size').value)||20;
  const color=document.getElementById('text-color-sel').value;
  const tl=document.getElementById('text-layer');
  const node=document.createElement('div');
  node.className='text-node';node.textContent=content;
  node.style.color=color;node.style.fontSize=size+'px';
  node.style.left='10%';node.style.top='10%';
  let startX,startY,origX,origY;
  const getPos=e=>e.touches?{x:e.touches[0].clientX,y:e.touches[0].clientY}:{x:e.clientX,y:e.clientY};
  node.addEventListener('touchstart',e=>{e.stopPropagation();selectTextNode(node);const p=getPos(e);startX=p.x;startY=p.y;origX=node.offsetLeft;origY=node.offsetTop;},{passive:true});
  node.addEventListener('touchmove',e=>{e.stopPropagation();e.preventDefault();const p=getPos(e);node.style.left=(origX+p.x-startX)+'px';node.style.top=(origY+p.y-startY)+'px';},{passive:false});
  node.addEventListener('mousedown',e=>{
    e.stopPropagation();selectTextNode(node);startX=e.clientX;startY=e.clientY;origX=node.offsetLeft;origY=node.offsetTop;
    const mm=ev=>{node.style.left=(origX+ev.clientX-startX)+'px';node.style.top=(origY+ev.clientY-startY)+'px';};
    const mu=()=>{document.removeEventListener('mousemove',mm);document.removeEventListener('mouseup',mu);};
    document.addEventListener('mousemove',mm);document.addEventListener('mouseup',mu);
  });
  tl.appendChild(node);textNodes.push(node);
  document.getElementById('text-content').value='';
  toast('テキスト配置（ドラッグで移動）');
}
function selectTextNode(node){if(selectedTextNode)selectedTextNode.classList.remove('selected');selectedTextNode=node;node.classList.add('selected');}
function clearTexts(){document.getElementById('text-layer').innerHTML='';textNodes=[];selectedTextNode=null;}

function flattenTexts(){
  if(!canvas.width)return;
  const tl=document.getElementById('text-layer');
  const cr=canvas.getBoundingClientRect();
  const sx=canvas.width/cr.width,sy=canvas.height/cr.height;
  textNodes.forEach(node=>{
    const nr=node.getBoundingClientRect();
    const x=(nr.left-cr.left)*sx,y=(nr.top-cr.top)*sy;
    const size=parseInt(node.style.fontSize)||20;
    ctx.font=`900 ${size*sx}px 'Noto Sans JP',sans-serif`;
    ctx.fillStyle=node.style.color;ctx.textBaseline='top';
    ctx.shadowColor='rgba(0,0,0,0.8)';ctx.shadowBlur=4;
    ctx.fillText(node.textContent,x,y);ctx.shadowBlur=0;
  });
  clearTexts();saveHistory();
}

function getCompressedDataURL(q){
  flattenTexts();
  const MAX=1200;const tmp=document.createElement('canvas');
  let w=canvas.width,h=canvas.height;
  if(w>MAX||h>MAX){if(w>h){h=Math.round(h*MAX/w);w=MAX;}else{w=Math.round(w*MAX/h);h=MAX;}}
  tmp.width=w;tmp.height=h;tmp.getContext('2d').drawImage(canvas,0,0,w,h);
  return tmp.toDataURL('image/jpeg',q||0.85);
}

async function exportPDF(){
  if(selectedOutputImages.length===0){toast('画像を選択してください');return;}
  toast('PDF生成中…');
  try{
    const{jsPDF}=window.jspdf;
    const doc=new jsPDF({orientation:'portrait',unit:'mm',format:'a4'});
    const memo=document.getElementById('memo-input').value.trim();
    const pageW=210,margin=15,imgW=210-margin*2;
    selectedOutputImages.forEach((item,idx)=>{
      if(idx>0)doc.addPage();
      doc.setFontSize(10);doc.setTextColor(100);
      doc.text('現場ツール PRO — '+item.label+' — '+new Date().toLocaleDateString('ja-JP'),margin,margin-4);
      const img=new Image();img.src=item.dataURL;
      const aspect=img.naturalHeight/img.naturalWidth;
      const imgH=Math.min(imgW*aspect,297-margin*2-20);
      doc.addImage(item.dataURL,'JPEG',margin,margin,imgW,imgH);
      if(memo){doc.setFontSize(10);doc.setTextColor(60);doc.text('メモ: '+memo,margin,margin+imgH+8);}
      doc.setFontSize(8);doc.setTextColor(150);doc.text(new Date().toLocaleDateString('ja-JP'),margin,290);
    });
    doc.save('現場記録_'+new Date().toLocaleDateString('ja-JP').replace(/\//g,'-')+'.pdf');
    toast('📄 PDFを保存しました');
  }catch(e){toast('PDF生成エラー: '+e.message);}
}
